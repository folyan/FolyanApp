    // Монетка: смена цифры происходит СТРОГО в момент прохождения ребра (scaleX близко к 0)
    private fun LinearLayout.addCoin() {
        val size = dp(150)
        val card = MaterialCardView(this@MainActivity).apply {
            radius = (size / 2).toFloat()
            strokeWidth = dp(2)
            strokeColor = Color.parseColor("#4A5568")
            cardElevation = dp(4).toFloat()
            layoutParams = LinearLayout.LayoutParams(size, size).apply {
                gravity = Gravity.CENTER_HORIZONTAL
                setMargins(0, dp(64), 0, dp(40))
            }
        }
        val text = TextView(this@MainActivity).apply {
            this.text = "0"
            textSize = 60f
            typeface = uiFont
            gravity = Gravity.CENTER
            layoutParams = FrameLayout.LayoutParams(FrameLayout.LayoutParams.MATCH_PARENT, FrameLayout.LayoutParams.MATCH_PARENT)
        }
        card.addView(text)
        addView(card)

        var currentSide = 0
        var isTossing = false

        card.setOnClickListener {
            if (isTossing) return@setOnClickListener
            isTossing = true

            val targetSide = Random.nextInt(2) // 0 или 1
            val tossAnimator = ValueAnimator.ofFloat(0f, 8f).apply {
                duration = 900
                interpolator = DecelerateInterpolator()
                var lastCycle = -1
                var switchedInCurrentCycle = false

                addUpdateListener { anim ->
                    val progress = anim.animatedValue as Float
                    val currentCycle = progress.toInt()
                    val cycleProgress = progress - currentCycle
                    val s = kotlin.math.abs(kotlin.math.cos(cycleProgress * Math.PI)).toFloat()
                    card.scaleX = s

                    if (currentCycle != lastCycle) {
                        lastCycle = currentCycle
                        switchedInCurrentCycle = false
                    }

                    // Меняем цифру строго в мертвой точке (когда монетка ребром s < 0.15)
                    if (s < 0.15f && !switchedInCurrentCycle) {
                        // На финальном цикле принудительно ставим целевую сторону
                        currentSide = if (currentCycle >= 7) targetSide else if (currentSide == 0) 1 else 0
                        text.text = currentSide.toString()
                        switchedInCurrentCycle = true
                    }
                }

                addListener(object : AnimatorListenerAdapter() {
                    override fun onAnimationEnd(animation: Animator) {
                        card.scaleX = 1f
                        currentSide = targetSide
                        text.text = currentSide.toString()
                        isTossing = false
                    }
                })
            }
            tossAnimator.start()
        }
    }

    // Возвращаем фокус и клавиатуру в поле, чтобы можно было писать сразу
    private fun keepFocus(input: EditText) {
        input.requestFocus()

    private fun sendInterp(input: EditText) {
        val text = input.text.toString().trim()
        val features = py.getModule("features")
        val code = interpCode

        if (features.callAttr("interp_is_running").toBoolean()) {
            if (text.isNotEmpty()) {
                bubble(text, right = true)
                features.callAttr("interp_answer", text)
                input.setText("")
            }
        } else if (code != null) {
            interpChat?.removeAllViews()
            bubble(interpName, right = true)
            if (text.isNotEmpty()) bubble(text, right = true)
            input.setText("")
            features.callAttr("interp_start", code, true, text.ifEmpty { null })
        } else if (text.isNotEmpty()) {
            bubble(text, right = true)
            input.setText("")
            features.callAttr("interp_start", text, false, null)
        }
        keepFocus(input)
    }

    private fun pickInterpFile() {
        pickHandler = { uri ->
            worker.execute {
                val name = displayName(uri)
                val text = try {
                    contentResolver.openInputStream(uri)!!.bufferedReader().use { it.readText() }
                } catch (e: Exception) {
                    null
                }
                main.post {
                    if (text != null) {
                        interpName = name
                        interpCode = text
                        refreshInterpFile()
                    } else {
                        bubble("Не удалось открыть файл", right = false)
                    }
                }
            }
        }
        picker.launch(arrayOf("*/*"))
    }

    private fun displayName(uri: Uri): String =
        contentResolver.query(uri, null, null, null, null)?.use { c ->
            val i = c.getColumnIndex(OpenableColumns.DISPLAY_NAME)
            if (c.moveToFirst() && i >= 0) c.getString(i) else null
        } ?: "файл.py"

    private fun clearInterpFile() {
        interpCode = null
        interpName = ""
        py.getModule("features").callAttr("interp_reset")
        refreshInterpFile()
    }

    private fun showPasswords() {
        val saved = loadState("passwords.json")
        setScreen(screen("Менеджер паролей") {
            val salt = field("Соль", saved.optString("salt", "FolyanApp"))
            val master = field("Мастер-пароль", saved.optString("master", ""), password = true)
            val site = field("Сайт", saved.optString("site", ""))
            val iters = field("Итерации", saved.optString("iterations", "500000"))
            val rememberFields = check("Запоминать соль, сайт и итерации", saved.has("salt"))
            val rememberMaster = check("Запоминать мастер-пароль", saved.has("master"))
            val out = label("")
            button("Сгенерировать") {
                val n = iters.text.toString().toIntOrNull() ?: 0
                val m = master.text.toString()
                val s = salt.text.toString()
                val t = site.text.toString()
                if (n < 1) {
                    out.text = "Итерации: целое число от 1"
                    return@button
                }
                val data = JSONObject()
                if (rememberFields.isChecked) {
                    data.put("salt", s)
                    data.put("site", t)
                    data.put("iterations", n.toString())
                }
                if (rememberMaster.isChecked) data.put("master", m)
                busy(out, "Считаю") {
                    py.getModule("state").callAttr("save", "passwords.json", stateDir, data.toString())
                    py.getModule("features").callAttr("password", m, s, t, n).toString()
                }
            }
        })
    }

import android.app.Activity
import android.os.Handler
import android.os.Looper
import com.chaquo.python.PyException
import com.chaquo.python.Python
import io.flutter.embedding.engine.FlutterEngine
import io.flutter.plugin.common.MethodChannel
import java.util.concurrent.Executors

// Мост Flutter <-> Python (Chaquopy). Канал: "folyan/download".
// Тяжёлые вызовы идут в отдельном потоке, ответ возвращается на главный поток.
class DownloadBridge(private val activity: Activity, engine: FlutterEngine) {
    private val worker = Executors.newSingleThreadExecutor()
    private val main = Handler(Looper.getMainLooper())

    // ffmpeg лежит рядом с нативными библиотеками как libffmpeg.so
    private val ffmpegPath: String =
        activity.applicationInfo.nativeLibraryDir + "/libffmpeg.so"

    private fun dl() = Python.getInstance().getModule("dl")

    init {
        MethodChannel(engine.dartExecutor.binaryMessenger, "folyan/download")
            .setMethodCallHandler { call, result ->
                when (call.method) {
                    "download" -> {
                        val url = call.argument<String>("url")!!
                        val height = call.argument<Int>("height")!!
                        val codec = call.argument<String>("codec")!!
                        val ai = call.argument<Boolean>("ai")!!
                        val fmt = call.argument<String>("fmt")!!
                        val outDir = call.argument<String>("outDir")!!
                        async(result) {
                            dl().callAttr("download", url, height, codec, ai, fmt, outDir, ffmpegPath)
                                .toString()
                        }
                    }
                    "convert" -> {
                        val src = call.argument<String>("src")!!
                        val fmt = call.argument<String>("fmt")!!
                        val outDir = call.argument<String>("outDir")!!
                        async(result) {
                            dl().callAttr("convert", src, fmt, outDir, ffmpegPath).toString()
                        }
                    }
                    "merge" -> {
                        val video = call.argument<String>("video")!!
                        val audio = call.argument<String>("audio")!!
                        val fmt = call.argument<String>("fmt")!!
                        val outDir = call.argument<String>("outDir")!!
                        async(result) {
                            dl().callAttr("merge", video, audio, outDir, fmt, ffmpegPath).toString()
                        }
                    }
                    "status" -> result.success(dl().callAttr("status").toString())
                    "cancel" -> {
                        dl().callAttr("cancel")
                        result.success(null)
                    }
                    else -> result.notImplemented()
                }
            }
    }

    private fun async(result: MethodChannel.Result, block: () -> String) {
        worker.execute {
            try {
                val value = block()
                main.post { result.success(value) }
            } catch (e: PyException) {
                // type: имя исключения Python, message: текст ошибки
                main.post { result.error("DL_FAILED", e.message, e.type) }
            }
        }
    }
}

const std = @import("std");

// Zero-width steganography, same format as main.zig:
//   each secret byte -> 8 bits, MSB first
//   bit 0 -> U+200B (E2 80 8B), bit 1 -> U+200C (E2 80 8C)
//   the hidden bits are appended after the cover text.
//
// Return codes:
//   >= 0  number of bytes written
//   -1    arithmetic overflow (lengths too large)
//   -2    output buffer too small

const bytes_per_secret_byte = 24;
const zw0 = [3]u8{ 0xE2, 0x80, 0x8B };
const zw1 = [3]u8{ 0xE2, 0x80, 0x8C };

/// How many secret bytes fit after `cover_len` bytes of cover text in `out_cap` bytes.
export fn folyan_stego_max_secret(cover_len: usize, out_cap: usize) usize {
    if (cover_len >= out_cap) return 0;
    return (out_cap - cover_len) / bytes_per_secret_byte;
}

/// Writes cover text followed by the hidden secret into `out`.
export fn folyan_stego_encode(
    cover: [*]const u8,
    cover_len: usize,
    secret: [*]const u8,
    secret_len: usize,
    out: [*]u8,
    out_cap: usize,
) isize {
    const hidden = std.math.mul(usize, secret_len, bytes_per_secret_byte) catch return -1;
    const need = std.math.add(usize, cover_len, hidden) catch return -1;
    if (need > out_cap) return -2;

    @memcpy(out[0..cover_len], cover[0..cover_len]);
    var pos: usize = cover_len;

    for (secret[0..secret_len]) |byte| {
        var bit: u3 = 7;
        while (true) : (bit -= 1) {
            const z = if (((byte >> bit) & 1) == 1) zw1 else zw0;
            @memcpy(out[pos..][0..3], &z);
            pos += 3;
            if (bit == 0) break;
        }
    }
    return @intCast(pos);
}

/// Extracts the hidden bytes from `input` into `out`.
export fn folyan_stego_decode(
    input: [*]const u8,
    in_len: usize,
    out: [*]u8,
    out_cap: usize,
) isize {
    const s = input[0..in_len];
    var acc: u8 = 0;
    var bits: u4 = 0;
    var written: usize = 0;
    var i: usize = 0;

    while (i + 3 <= s.len) {
        if (s[i] == 0xE2 and s[i + 1] == 0x80 and (s[i + 2] == 0x8B or s[i + 2] == 0x8C)) {
            const bit: u8 = if (s[i + 2] == 0x8C) 1 else 0;
            acc = (acc << 1) | bit;
            bits += 1;
            if (bits == 8) {
                if (written >= out_cap) return -2;
                out[written] = acc;
                written += 1;
                bits = 0;
                acc = 0;
            }
            i += 3;
        } else {
            i += 1;
        }
    }
    return @intCast(written);
}

        self.len = 0;

        self.items[self.len] = s;
        self.len += 1;

/// Mirrors convert_media() in nativebins.py. Attempts 0 and 1.
/// The caller prepends "-hide_banner -nostdin -y" as run_ffmpeg() does.
pub fn convertArgs(args: *Args, src: []const u8, out: []const u8, fmt: []const u8, attempt: usize) bool {
    args.reset();

    if (isAudio(fmt)) {
        switch (attempt) {
            0 => {
                args.add(&.{ "-i", src, "-vn", "-map", "0:a:0" });
                args.add(audioCodecArgs(fmt) orelse return false);
                args.push(out);
            },
            1 => {
                args.add(&.{ "-i", src, "-vn" });
                args.push(out);
            },
            else => return false,
        }
        return true;
    }

    const flags: []const []const u8 = if (std.mem.eql(u8, fmt, "mp4"))
        &.{ "-movflags", "+faststart" }
    else
        &.{};
    const maps = [_][]const u8{ "-map", "0:v:0", "-map", "0:a?", "-sn" };

    switch (attempt) {
        0 => {
            args.add(&.{ "-i", src });
            args.add(&maps);
            args.add(&.{ "-c", "copy" });
            args.add(flags);
            args.push(out);
        },
        1 => {
            args.add(&.{ "-i", src });
            args.add(&maps);
            args.add(&.{ "-c:v", "libx264", "-preset", "veryfast", "-crf", "23", "-c:a", "aac", "-b:a", "192k" });
            args.add(flags);
            args.push(out);
        },
        else => return false,
    }
    return true;
}

/// Mirrors merge_media() in nativebins.py. Attempts 0, 1 and 2.
pub fn mergeArgs(args: *Args, video: []const u8, audio: []const u8, out: []const u8, fmt: []const u8, attempt: usize) bool {
    args.reset();

    const flags: []const []const u8 = if (std.mem.eql(u8, fmt, "mp4"))
        &.{ "-movflags", "+faststart" }
    else
        &.{};

    args.add(&.{ "-i", video, "-i", audio, "-map", "0:v:0", "-map", "1:a:0", "-shortest" });
    switch (attempt) {
        0 => args.add(&.{ "-c", "copy" }),
        1 => args.add(&.{ "-c:v", "copy", "-c:a", "aac", "-b:a", "192k" }),
        2 => args.add(&.{ "-c:v", "libx264", "-preset", "veryfast", "-crf", "23", "-c:a", "aac", "-b:a", "192k" }),
        else => return false,
    }
    args.add(flags);
    args.push(out);
    return true;
}

/// Parses "H:MM:SS(.ff)" into seconds.
pub fn parseHms(s: []const u8) ?f64 {
    var it = std.mem.splitScalar(u8, s, ':');
    const h = it.next() orelse return null;
    const m = it.next() orelse return null;
    const sec = it.next() orelse return null;
    if (it.next() != null) return null;

    const hh = std.fmt.parseInt(u32, h, 10) catch return null;
    const mm = std.fmt.parseInt(u32, m, 10) catch return null;
    const ss = std.fmt.parseFloat(f64, sec) catch return null;
    return @as(f64, @floatFromInt(hh)) * 3600.0 + @as(f64, @floatFromInt(mm)) * 60.0 + ss;
}

/// Finds `key` (e.g. "Duration:" or "time=") in an ffmpeg line and parses the time after it.
pub fn findTime(line: []const u8, key: []const u8) ?f64 {
    const start = std.mem.indexOf(u8, line, key) orelse return null;
    const rest = std.mem.trimStart(u8, line[start + key.len ..], " ");
    var end: usize = 0;
    while (end < rest.len and (std.ascii.isDigit(rest[end]) or rest[end] == ':' or rest[end] == '.')) {
        end += 1;
    }
    return parseHms(rest[0..end]);
}

/// Progress percent as in run_ffmpeg(): capped at 99 while running.
pub fn percent(current: f64, total: f64) u8 {
    if (total <= 0) return 0;
    const p = @floor(current * 100.0 / total);
    if (p >= 99) return 99;
    if (p < 0) return 0;
    return @intFromFloat(p);
}

/// Candidate name for attempt `index`: index 0 gives "base.ext",
/// index n gives "base (n).ext". Walk indices from 0 and use the first
/// name whose file does not exist yet. Merge output uses base "<name>_merged".
pub fn candidateName(buf: []u8, base: []const u8, ext: []const u8, index: usize) std.fmt.BufPrintError![]u8 {
    if (index == 0) return std.fmt.bufPrint(buf, "{s}.{s}", .{ base, ext });
    return std.fmt.bufPrint(buf, "{s} ({d}).{s}", .{ base, index, ext });
}

// ---------------------------------------------------------------------------
// Running ffmpeg (std.process, same calls as process.zig's run())

pub const Error = error{ Interrupted, FfmpegFailed, ArgsTooLong } || process.SpawnError || Io.Cancelable || Io.UnexpectedError;

/// Callbacks supplied by the caller (UI, JNI, dart:ffi, ...).
pub const Hooks = struct {
    ctx: *anyopaque,
    /// Called with 0..99 while encoding.
    onProgress: *const fn (ctx: *anyopaque, percent: u8) void,
    /// Polled between reads. Return true to stop the conversion.
    isCancelled: *const fn (ctx: *anyopaque) bool,
    /// Removes a partial output file after a failed attempt or cancel.
    removeFile: *const fn (ctx: *anyopaque, path: []const u8) void,
};

const line_capacity = 2048;

const LineState = struct {
    total: ?f64 = null,
    hooks: Hooks,

    fn handle(self: *LineState, line: []const u8) void {
        if (self.total == null) {
            self.total = findTime(line, "Duration:");
        }
        const cur = findTime(line, "time=") orelse return;
        const total = self.total orelse return;
        self.hooks.onProgress(self.hooks.ctx, percent(cur, total));
    }
};

/// Spawns `ffmpeg -hide_banner -nostdin -y <args>` and waits for it.
fn runOnce(io: Io, exe: []const u8, args: []const []const u8, hooks: Hooks) Error!void {
    var argv: Args = .{};
    argv.push(exe);
    argv.push("-hide_banner");
    argv.push("-nostdin");
    argv.push("-y");
    argv.add(args);

    // argv[0] is the absolute path to ffmpeg; the default exe = .detect uses it as-is.
    var child = try process.spawn(io, .{
        .argv = argv.slice(),
        .stdin = .ignore,
        .stdout = .ignore,
        .stderr = .pipe,
    });
    defer child.kill(io);

    var state = LineState{ .hooks = hooks };
    var line: [line_capacity]u8 = undefined;
    var line_len: usize = 0;

    var rbuf: [4096]u8 = undefined;
    var file_reader = child.stderr.?.reader(io, &rbuf);
    var chunk: [1024]u8 = undefined;

    while (true) {
        if (hooks.isCancelled(hooks.ctx)) {
            child.kill(io);
            return error.Interrupted;
        }
        // The real cause is in file_reader.err; the caller only needs "failed".
        const n = file_reader.interface.readSliceShort(&chunk) catch return error.FfmpegFailed;
        if (n == 0) break;

        // ffmpeg separates progress updates with '\r', messages with '\n'.
        for (chunk[0..n]) |c| {
            if (c == '\r' or c == '\n') {
                state.handle(line[0..line_len]);
                line_len = 0;
            } else if (line_len < line.len) {
                line[line_len] = c;
                line_len += 1;
            }
        }
    }
    if (line_len > 0) state.handle(line[0..line_len]);

    const term = try child.wait(io);
    switch (term) {
        .exited => |code| if (code != 0) return error.FfmpegFailed,
        else => return error.FfmpegFailed,
    }
}

/// Runs every fallback attempt from `build` until one succeeds.
/// On failure the partial output is removed; on cancel the error is Interrupted.
fn runAttempts(
    io: Io,
    exe: []const u8,
    out: []const u8,
    hooks: Hooks,
    comptime build: anytype,
    build_ctx: anytype,
) Error!void {
    var args: Args = .{};
    var last_err: Error = error.FfmpegFailed;
    var attempt: usize = 0;

    while (build(&args, build_ctx, attempt)) : (attempt += 1) {
        runOnce(io, exe, args.slice(), hooks) catch |err| switch (err) {
            error.Interrupted => {
                hooks.removeFile(hooks.ctx, out);
                return error.Interrupted;
            },
            else => {
                last_err = err;
                hooks.removeFile(hooks.ctx, out);
                continue;
            },
        };
        return;
    }
    return last_err;
}

const ConvertCtx = struct { src: []const u8, out: []const u8, fmt: []const u8 };

fn buildConvert(args: *Args, ctx: ConvertCtx, attempt: usize) bool {
    return convertArgs(args, ctx.src, ctx.out, ctx.fmt, attempt);
}

const MergeCtx = struct { video: []const u8, audio: []const u8, out: []const u8, fmt: []const u8 };

fn buildMerge(args: *Args, ctx: MergeCtx, attempt: usize) bool {
    return mergeArgs(args, ctx.video, ctx.audio, ctx.out, ctx.fmt, attempt);
}

/// Converts `src` to `out` (a unique path chosen by the caller).
/// Mirrors convert_media() + run_attempts().
pub fn convertFile(io: Io, exe: []const u8, src: []const u8, out: []const u8, fmt: []const u8, hooks: Hooks) Error!void {
    return runAttempts(io, exe, out, hooks, buildConvert, ConvertCtx{ .src = src, .out = out, .fmt = fmt });
}

/// Merges `video` with the external audio `audio` into `out` (unique path from caller).
/// Mirrors merge_media() + run_attempts().
pub fn mergeFile(io: Io, exe: []const u8, video: []const u8, audio: []const u8, out: []const u8, fmt: []const u8, hooks: Hooks) Error!void {
    return runAttempts(io, exe, out, hooks, buildMerge, MergeCtx{ .video = video, .audio = audio, .out = out, .fmt = fmt });
}

const std = @import("std");

const Hmac = std.crypto.auth.hmac.sha2.HmacSha256;
const b64 = std.base64.standard.Encoder;

pub const default_salt = "FolyanApp";
pub const default_iterations: u32 = 500_000;

/// Derived key length (SHA-256 output).
const dk_len = 32;
/// base64 of 32 bytes, with one '=' pad.
const b64_len = b64.calcSize(dk_len);
/// Output: first 30 chars of base64 (with '/' -> '_') plus "_!".
pub const password_len = 30 + 2;

pub const Error = error{ WeakParameters, OutOfMemory, OutputTooLong };

/// Mirrors passwordmanager.generate_password:
///   pbkdf2_hmac(sha256, master, salt ++ site, iterations, 32)
///   -> base64 -> '/' replaced by '_' -> first 30 chars + "_!"
pub fn generatePassword(
    allocator: std.mem.Allocator,
    out: *[password_len]u8,
    master: []const u8,
    salt: []const u8,
    site: []const u8,
    iterations: u32,
) Error!void {
    if (iterations < 1) return error.WeakParameters;

    const pbkdf2_salt = try allocator.alloc(u8, salt.len + site.len);
    defer allocator.free(pbkdf2_salt);
    @memcpy(pbkdf2_salt[0..salt.len], salt);
    @memcpy(pbkdf2_salt[salt.len..], site);

    var key: [dk_len]u8 = undefined;
    std.crypto.pwhash.pbkdf2(&key, master, pbkdf2_salt, iterations, Hmac) catch |err| switch (err) {
        error.WeakParameters => return error.WeakParameters,
        error.OutputTooLong => return error.OutputTooLong,
    };

    var encoded: [b64_len]u8 = undefined;
    _ = b64.encode(&encoded, &key);

    for (encoded[0..30], 0..) |c, i| {
        out[i] = if (c == '/') '_' else c;
    }
    out[30] = '_';
    out[31] = '!';
}

//
// Returns 0 on success, negative on error. `out` must point to 32 writable bytes.
// `iterations` = 0 is rejected (same as Python's "Итерации: целое число от 1").
// ---------------------------------------------------------------------------
export fn folyan_generate_password(
    master_ptr: [*]const u8,
    master_len: usize,
    salt_ptr: [*]const u8,
    salt_len: usize,
    site_ptr: [*]const u8,
    site_len: usize,
    iterations: u32,
    out: [*]u8,
) c_int {
    var result: [password_len]u8 = undefined;
    generatePassword(
        std.heap.page_allocator,
        &result,
        master_ptr[0..master_len],
        salt_ptr[0..salt_len],
        site_ptr[0..site_len],
        iterations,
    ) catch |err| return switch (err) {
        error.WeakParameters => -1,
        error.OutputTooLong => -2,
        error.OutOfMemory => -3,
    };
    @memcpy(out[0..password_len], &result);
    return 0;
}

// One Io instance per library, created on first use. It must outlive every
// call that uses it, so it lives in a global.
// Not thread-safe on the very first call: call folyan_init() from Dart before

    if (!threaded_ready) {

    }

    io_global.deinit();
}

fn adapter(c: *const CCallbacks) conv.Hooks {
    return .{
        .ctx = @constCast(@ptrCast(c)),
        .onProgress = struct {
            fn f(ctx: *anyopaque, pct: u8) void {
                const cb: *const CCallbacks = @ptrCast(@alignCast(ctx));
                if (cb.on_progress) |p| p(cb.ctx, pct);
            }
        }.f,
        .isCancelled = struct {
            fn f(ctx: *anyopaque) bool {
                const cb: *const CCallbacks = @ptrCast(@alignCast(ctx));
                if (cb.is_cancelled) |p| return p(cb.ctx);
                return false;
            }
        }.f,
        .removeFile = struct {
            fn f(ctx: *anyopaque, path: []const u8) void {
                const cb: *const CCallbacks = @ptrCast(@alignCast(ctx));
                var z: [path_capacity:0]u8 = undefined;
                if (path.len >= z.len) return;
                @memcpy(z[0..path.len], path);
                z[path.len] = 0;
                if (cb.remove_file) |p| p(cb.ctx, &z);
            }
        }.f,
    };
}

/// Picks "<dir>/<base>.<ext>" or "<dir>/<base> (n).<ext>" that does not exist yet.
fn pickOutput(
    cb: *const CCallbacks,
    buf: []u8,
    dir: []const u8,
    base: []const u8,
    ext: []const u8,
) ?[]const u8 {
    var name_buf: [path_capacity]u8 = undefined;
    var i: usize = 0;
    while (i < max_tries) : (i += 1) {
        const name = conv.candidateName(&name_buf, base, ext, i) catch return null;
        const full = std.fmt.bufPrint(buf, "{s}/{s}", .{ dir, name }) catch return null;
        if (full.len >= buf.len) return null;
        buf[full.len] = 0;
        const z: [*:0]const u8 = @ptrCast(buf.ptr);
        if (!cb.exists(cb.ctx, z)) return buf[0..full.len];
    }
    return null;
}

fn mapError(err: anyerror) c_int {
    return switch (err) {
        error.Interrupted => -3,
        error.FfmpegFailed => -4,
        else => -5,
    };
}

fn copyOut(path: []const u8, out_path: [*]u8, out_cap: usize) c_int {
    if (path.len + 1 > out_cap) return -1;
    @memcpy(out_path[0..path.len], path);
    out_path[path.len] = 0;
    return 0;
}

/// Converts src into a new file in out_dir with extension `fmt`.
export fn folyan_convert(
    exe: [*:0]const u8,
    src: [*:0]const u8,
    out_dir: [*:0]const u8,
    fmt: [*:0]const u8,
    callbacks: *const CCallbacks,
    out_path: [*]u8,
    out_cap: usize,
) c_int {
    const src_s = std.mem.span(src);
    const dir_s = std.mem.span(out_dir);
    const fmt_s = std.mem.span(fmt);
    if (fmt_s.len == 0) return -1;

    const base = std.fs.path.stem(std.fs.path.basename(src_s));
    var buf: [path_capacity]u8 = undefined;
    const out = pickOutput(callbacks, &buf, dir_s, base, fmt_s) orelse return -2;

    conv.convertFile(ioForLibrary(), std.mem.span(exe), src_s, out, fmt_s, adapter(callbacks)) catch |err| {
        return mapError(err);
    };
    return copyOut(out, out_path, out_cap);
}

/// Merges the video `video` with the external audio `audio` into out_dir.
/// Output is "<video base>_merged.<fmt>".
export fn folyan_merge(
    exe: [*:0]const u8,
    video: [*:0]const u8,
    audio: [*:0]const u8,
    out_dir: [*:0]const u8,
    fmt: [*:0]const u8,
    callbacks: *const CCallbacks,
    out_path: [*]u8,
    out_cap: usize,
) c_int {
    const video_s = std.mem.span(video);
    const audio_s = std.mem.span(audio);
    const dir_s = std.mem.span(out_dir);
    const fmt_s = std.mem.span(fmt);
    if (fmt_s.len == 0) return -1;

    const base = std.fs.path.stem(std.fs.path.basename(video_s));
    var merged_base: [path_capacity]u8 = undefined;
    const merged = std.fmt.bufPrint(&merged_base, "{s}_merged", .{base}) catch return -1;

    var buf: [path_capacity]u8 = undefined;
    const out = pickOutput(callbacks, &buf, dir_s, merged, fmt_s) orelse return -2;

    conv.mergeFile(ioForLibrary(), std.mem.span(exe), video_s, audio_s, out, fmt_s, adapter(callbacks)) catch |err| {
        return mapError(err);
    };
    return copyOut(out, out_path, out_cap);
}
